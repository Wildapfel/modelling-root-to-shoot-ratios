	\section{Data Exploration}

		\subsection*{Data Distribution}
	
			\begin{figure}[h!]
				\centering
				\includegraphics[scale=0.5]{"\imgDir/hists_sprout_mass_root_mas_mass_ratio.png"}
				\caption{Exploitative analysis. Left: Sprout mass data distribution.  Middle: Root mass data distribution.  Right: Mass-Ratio distribution. }
			\end{figure}
			
			\textbf{Masses}
			\begin{itemize}
				\reduceVerticalSpace
				\reduceItemSpace
				\item The masses of sprouts and roots tend to be slightly right skewed (rechtsschief)  
				\item $Weights \in \mathbb{R}_+$   
				\item Just a normal distribution might now be appropriate, because its highly likely it will extend into the negatives and doesn't catch the skewness  
				\item A Gamma might fits the marginal distribution better, because: $X \in \mathbb{R}_{+}$
			\end{itemize}
		
			\textbf{Ratio}
			\begin{itemize}
				\reduceVerticalSpace
				\reduceItemSpace
				\item The skweness seem to not be present here   
				\item $Ratio \in [0, 1]$    
				\item Intuitivly (and by their definition) a beta might fit best here  
			\end{itemize}
		
		\subsection*{Data Correlation}
			\begin{figure}[h!]
				\centering
				\includegraphics[scale=0.5]{"\imgDir/corr_root_sprout_mass.png"}
				\caption{Scatter plot of root and sprout mass.}
			\end{figure}
			
			\begin{itemize}
				\item Hight correlation between the masses (r=$\approx$0.76)
				\item Implying a statistical dependency between both parameters
			\end{itemize}

	\section{Ratio-Modelling using Beta Distribution}
	
		\subsection*{Root-To-Shoot Ratio}
			
			\textbf{Theoretically}
			
			\begin{tabular}{ll}
				$\quad X_1 \sim Ga(x_1| \alpha_1, \beta)$ & Shown in previous section \\
				$\quad X_2 \sim Ga(x_2| \alpha_2, \beta)$ & Shown in previous section \\
				$\quad Y = \frac{X_1}{X_1 + X_2}$ & Transformation \\
				$\quad Y \sim Be(y|\alpha_1, \alpha_2)$ & Under the assumption that $\beta_{root} = \beta_{sprout} = \beta$ and $X_1 \perp\!\!\!\perp X_2 $  (independent). \\ 
			\end{tabular}

	\section{Interpretation}
		\begin{itemize}
			\reduceVerticalSpace
			\reduceItemSpace
			\item The poor fit quality of the estimated beta distribution  is an indicator that both masses are statistically dependent and further supports the correlation 
			\item This validates, that the correlation breaks the independency assumption of the theoretical beta
			\item One could interpret that this peak is the result of stronger dependency (also correlation) of those seedlings masses
			\item Unfortunately, I don't have to much information about the data, especially I don't know if certain seedlings perform better than other (for breeding purpose)
			\item I could argue that this ratio range (0.5 - 0.7) might be a optimal target for breeding selection (with statistical power)
		\end{itemize}
